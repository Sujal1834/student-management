--------------create student table-------------------
CREATE TABLE students (
    id SERIAL PRIMARY KEY,
    name VARCHAR(100) NOT NULL,
    email VARCHAR(100) NOT NULL,
    roll_no VARCHAR(20) UNIQUE NOT NULL,
    dob DATE,
    gender VARCHAR(10),
    course_id INT REFERENCES courses(id) ON DELETE CASCADE,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

    CREATE OR REPLACE PROCEDURE create_course(
        name VARCHAR
    )
    LANGUAGE plpgsql
    AS $$
    BEGIN
        INSERT INTO courses(course_name) VALUES (name);
    END;
    $$;

    CREATE OR REPLACE PROCEDURE create_student(
        s_name VARCHAR,
        s_email VARCHAR,
        s_r_n VARCHAR,
        s_dob DATE,
        s_gender VARCHAR,
        s_course_id INT
    )
    LANGUAGE plpgsql
    AS $$
    BEGIN
        INSERT INTO students(roll_no,name,email,dob,gender,course_id) VALUES
        (s_r_n,s_name,s_email,s_dob,s_gender,s_course_id);
    END;
    $$;

    CALL create_student('sujal','sujal@gmail.com','1001','2004-03-18','Male',1);


    ...................update student......................

    CREATE OR REPLACE PROCEDURE update_student(
        s_id INT,
        s_name VARCHAR,
        s_email VARCHAR,
        s_roll_no VARCHAR,
        s_dob DATE,
        s_gender VARCHAR,
        s_course_id INT
    )
    LANGUAGE plpgsql
    AS $$
    BEGIN
        UPDATE students SET name=s_name,email=s_email,roll_no=s_roll_no,dob=s_dob,gender=s_gender,course_id=s_course_id 
        WHERE id = s_id;
    END;
    $$;

----------------------Functions-------------------------------

    ..................get courses...........................

    CREATE OR REPLACE FUNCTION get_courses()
    RETURNS TABLE(
        id INT,
        course_name VARCHAR
    )
    LANGUAGE plpgsql
    AS $$
    BEGIN
        RETURN QUERY SELECT c.id,c.course_name FROM courses c ORDER BY c.id;
    END;
    $$;

    SELECT * FROM get_courses();

    .................get students............................

    CREATE OR REPLACE FUNCTION get_students()
    RETURNS TABLE(
		id INT,
        roll_no VARCHAR,
        name VARCHAR,
        email VARCHAR,
        dob DATE,
        gender VARCHAR,
        course_id INT,
		password VARCHAR
    )
    LANGUAGE plpgsql
    AS $$
    BEGIN
        RETURN QUERY SELECT s.id,s.roll_no, s.name, s.email, s.dob, s.gender, s.course_id, s.password FROM students s WHERE is_faculty='False' ORDER BY s.roll_no;
    END;

    ..................update course................

    CREATE OR REPLACE FUNCTION update_course(
        c_id INT,
        c_name VARCHAR
    )
    RETURNS BOOLEAN
    LANGUAGE plpgsql
    AS $$
    BEGIN
        UPDATE courses SET course_name=c_name WHERE id=c_id;
        RETURN TRUE;
    END;
    $$;

    SELECT update_course(1,'mscit');

    ....................delete course.....................

    CREATE OR REPLACE FUNCTION delete_course(
        c_id INT
    )
    RETURNS TABLE(success BOOLEAN, message TEXT)
    LANGUAGE plpgsql
    AS $$
    BEGIN
        DELETE FROM courses WHERE id=c_id;

        IF NOT FOUND THEN
            RETURN QUERY
            SELECT FALSE, 'Course not found';
            RETURN;
        END IF;
        
        RETURN QUERY
        SELECT TRUE, 'Course deleted successfully';
    END;
    $$;

    SELECT delete_course(12);

    ..................search student by name..........


    CREATE OR REPLACE FUNCTION search_by_name(
        s_name VARCHAR
    )
    RETURNS TABLE(
        id INT,
        roll_no VARCHAR,
        name VARCHAR,
        email VARCHAR,
        dob DATE,
        gender VARCHAR,
        course_id INT
    )
    LANGUAGE plpgsql
    AS $$
    BEGIN
            RETURN QUERY SELECT s.id,s.roll_no, s.name, s.email, s.dob, s.gender, s.course_id FROM students s WHERE s.name ILIKE '%' || s_name || '%' ORDER BY s.roll_no;
    END;
    $$;

    ......................get student by email and password............

    CREATE OR REPLACE FUNCTION get_student_by_email(
        s_email VARCHAR,
        s_password VARCHAR
    )
    RETURNS TABLE(
        id INT,
        name VARCHAR,
        email VARCHAR,
        password VARCHAR,
		is_active BOOLEAN
    )
    LANGUAGE plpgsql
    AS $$
    BEGIN
        RETURN QUERY SELECT s.id,s.name,s.email,s.password,s.is_active FROM students s WHERE s.email = s_email AND s.password = s_password;
    END;
    $$;

    ........................create/update assignment............................
    CREATE OR REPLACE FUNCTION save_assignment(
        p_assignment_id INTEGER,
        p_title TEXT,
        p_description TEXT,
        p_due_date DATE,
        p_course_id INTEGER,
        p_student_id INTEGER,
        p_file TEXT 
    )
    RETURNS BOOLEAN
    LANGUAGE plpgsql
    AS $$
    BEGIN

        IF p_assignment_id IS NULL THEN
            INSERT INTO assignments
            (title, description, due_date, course_id, student_id, file)
            VALUES
            (p_title, p_description, p_due_date, p_course_id, p_student_id, p_file);

            RETURN true;
        END IF;

        IF p_file IS NOT NULL THEN
            UPDATE assignments
            SET
                title = p_title,
                description = p_description,
                due_date = p_due_date,
                course_id = p_course_id,
                file = p_file
            WHERE id = p_assignment_id;
        ELSE
            UPDATE assignments
            SET
                title = p_title,
                description = p_description,
                due_date = p_due_date,
                course_id = p_course_id
            WHERE id = p_assignment_id;
        END IF;

        RETURN true;
    END;
    $$;

    ...........................search student....................................
    CREATE OR REPLACE FUNCTION search_students(
        p_course_id INTEGER DEFAULT NULL,
        p_name TEXT DEFAULT NULL
    )
    RETURNS TABLE (
        roll_no VARCHAR,
        name VARCHAR,
        email VARCHAR,
        dob DATE,
        gender VARCHAR,
        course_id INTEGER
    )
    LANGUAGE plpgsql
    AS $$
    BEGIN
        RETURN QUERY
        SELECT
            s.roll_no,
            s.name,
            s.email,
            s.dob,
            s.gender,
            s.course_id
        FROM students s
        WHERE
            s.is_faculty = FALSE
            AND (p_course_id IS NULL OR s.course_id = p_course_id)
            AND (p_name IS NULL OR s.name ILIKE '%' || p_name || '%');
    END;
    $$;
